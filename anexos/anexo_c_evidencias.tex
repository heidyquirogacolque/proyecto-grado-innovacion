\begin{lstlisting}[language=SQL, caption={Verificación de consistencia de cupos (PostgreSQL), anonimizada.}, label={lst:verificacion_cupos}]
-- Verificación post-pruebas: saldo de cupos por especialidad (anonimizado)
SELECT nombre AS especialidad,
       cupos_maximos,
       cupos_disponibles,
       (cupos_maximos - cupos_disponibles) AS matriculados
FROM especialidad
ORDER BY nombre;
-- Resultado: 7 filas, cupos_disponibles >=0, 0 filas con sobrecupo
